// ─────────────────────────────────────────────────────────────────────────────
// N2V6.ino — PSA nitrogen generator controller
//
// Implements requirements v1.1. Monolithic sketch: the only #includes are
// Wire.h and the DFRobot Multigas library; the LED and LCD mini-libraries are
// pasted in below. Target: Arduino UNO R4 Minima / UNO R4 WiFi.
//
// Architecture, in one paragraph: loop() reads every input exactly once into
// `inputs` (a consistent view for this tick), then runs three state-machine
// controllers (Tower, Compressor, O2) which drive the hardware outputs and
// record their results in `outputs`, then renders both displays from
// `outputs`/`inputs` using change-only writes. There is no delay() after
// setup() returns; every timed behavior is a millis() deadline checked each
// tick. ADC is the 10-bit default (hardcoded — no analogReadResolution call).
// ─────────────────────────────────────────────────────────────────────────────
#include <Wire.h>
#include "DFRobot_MultiGasSensor.h"  // SEN0465; readGasConcentrationPPM() → %vol for O2

const bool runBIST = true;  // enable or disable the running of the Build In Self Test (BIST)

const uint8_t TBS_PIN = 0;  // INPUT_PULLUP, ON = LOW (maintained)
const uint8_t TOB_PIN = 1;  // INPUT_PULLUP, ON = LOW (momentary)

/* ── Digital outputs (LOW == OFF, HIGH == ON for all) ────────────────────── */
const uint8_t LEFT_TOWER_VALVE_PIN = 4;   // HIGH = open
const uint8_t RIGHT_TOWER_VALVE_PIN = 7;  // HIGH = open
const uint8_t O2_FLUSH_VALVE_PIN = 11;    // HIGH = open
const uint8_t SSR_PIN = 8;                // HIGH = compressor on

/* ── Analog inputs ───────────────────────────────────────────────────────── */
const uint8_t SUPPLY_PRESSURE_PIN = A0;  // 0–150 PSI air supply
const uint8_t LOW_N2_PIN = A3;           // 0–30  PSI low-pressure N2
const uint8_t HIGH_N2_PIN = A5;          // 0–150 PSI high-pressure N2

/* ── I2C addresses ───────────────────────────────────────────────────────── */
const uint8_t I2C_ADDR_DISP4 = 0x24;  // TM1650 control; digit regs at 0x34..0x37
const uint8_t I2C_ADDR_LCD = 0x27;    // 20x4 LCD via PCF8574
const uint8_t I2C_ADDR_O2 = 0x74;     // SEN0465 (SEL dip = 0)

/* ── Pressure scaling: 0.5–4.5 V ratiometric into a 10-bit ADC ───────────── */
const uint16_t RAW_MIN = 102;                // counts at 0.5 V (1023 * 0.5/5.0)
const uint16_t RAW_MAX = 921;                // counts at 4.5 V (1023 * 4.5/5.0)
const uint16_t AIR_FULLSCALE_X10 = 1500;     // 150.0 PSI, stored x10
const uint16_t N2LOW_FULLSCALE_X100 = 3000;  //  30.00 PSI, stored x100
const uint16_t N2HIGH_FULLSCALE_X10 = 1500;  // 150.0 PSI, stored x10

/* ── Tower controller params ─────────────────────────────────────────────── */
const uint16_t airLowOff = 900;         //  PSI x10   TODO: tune
const uint16_t airLowOn = 1200;         //  PSI x10   TODO: tune
const uint32_t towerFillTime = 59250;   // ms         TODO: tune
const uint32_t towerOverlapTime = 750;  // ms         TODO: tune

/* ── Compressor controller params ────────────────────────────────────────── */
const uint16_t n2LowOff = 1000;   //  PSI x100 TODO: tune
const uint16_t n2LowOn = 2000;    //  PSI x100 TODO: tune
const uint16_t n2HighOn = 1000;   //  PSI x10  TODO: tune
const uint16_t n2HighOff = 1200;  //  PSI x10  TODO: tune

/* ── O2 controller params ────────────────────────────────────────────────── */

const uint32_t O2SampleTime = 250;        // ms between samples
const uint8_t O2SampleCount = 10;         // samples averaged per cycle

// ─────────────────────────────────────────────────────────────────────────────
// TM1650 mini-library (minified from TM1650min.h v4)
//
// Theory: the TM1650 has no register addressing in the usual sense — the chip
// occupies I2C address 0x24 for its control byte and 0x34..0x37 for the four
// digit registers, one address per digit. Writing a digit is a one-byte
// transmission to that digit's address. The control byte packs brightness
// (bits 4-6) and display on/off (bit 0); brightness is fixed at HW level 1.
//
// ─────────────────────────────────────────────────────────────────────────────
const uint8_t SEG[16] = { 0x3F, 0x06, 0x5B, 0x4F, 0x66, 0x6D, 0x7D, 0x07,
                          0x7F, 0x6F, 0x77, 0x7C, 0x39, 0x5E, 0x79, 0x71 };
struct TM1650 {
  uint8_t digitCache[4] = { 0xFF, 0xFF, 0xFF, 0xFF };  // last byte written per digit

  // One-byte I2C write to an arbitrary address (control or digit register).
  void writeByte(uint8_t addr, uint8_t value) {
    Wire.beginTransmission(addr);
    Wire.write(value);
    Wire.endTransmission();
  }

  void begin() {
    delay(50);
    on();
  }  // power-up settle, then enable

  void on() {
    writeByte(I2C_ADDR_DISP4, 0x11);
  }  // brightness HW1 | display on
  
  void off() {
    writeByte(I2C_ADDR_DISP4, 0x10);
  }  // brightness HW1 | display off

  // Write one digit's segment byte, but only if it changed. This per-digit
  // cache is what makes unconditional per-tick display refresh affordable.
  void writeDigit(uint8_t digit, uint8_t segments) {
    if (digitCache[digit] != segments) {
      digitCache[digit] = segments;
      writeByte(0x34 + digit, segments);
    }
  }

  // Show a 0..9999 value; dotDigit (0..3) lights that digit's decimal point.
  void show(uint16_t value, int8_t dotDigit = -1) {
    for (int8_t i = 3; i >= 0; --i, value /= 10)
      writeDigit(i, SEG[value % 10] | (i == dotDigit ? 0x80 : 0));
  }

  // Show a 16-bit value as four HEX digits (used for BIST step numbers).
  void showHex(uint16_t value) {
    for (int8_t i = 3; i >= 0; --i, value >>= 4) writeDigit(i, SEG[value & 15]);
  }

  void blank() {
    for (uint8_t i = 0; i < 4; i++) writeDigit(i, 0);
  }

  // All-dashes (segment G), used while N2% has no valid reading.
  void dashes(int8_t dotDigit = -1) {
    for (uint8_t i = 0; i < 4; i++) writeDigit(i, 0x40 | (i == (uint8_t)dotDigit ? 0x80 : 0));
  }
};

// ─────────────────────────────────────────────────────────────────────────────
// 20x4 LCD / PCF8574 mini-library (minified from LCD20x4min.h v3)
//
// Theory: the PCF8574 is an 8-bit I2C latch wired to the HD44780 in 4-bit
// mode: bits 7..4 carry the data nibble, bit 0 = RS, bit 2 = EN, bit 3 =
// backlight. Each LCD byte is sent as two nibbles, each strobed by pulsing EN
// high then low (the LCD latches on the falling edge). The backlight bit must
// be OR'd into every latch write or the backlight blinks off mid-transfer.
//
// Added vs. v3: print(row, col, text) so display fields can be written at
// fixed positions. Dropped: the writeLine alias (use print(row, 0, text)).
// begin() and clear() use delay() — callable from setup()/BIST only.
// ─────────────────────────────────────────────────────────────────────────────
struct LCD20x4 {
  uint8_t backlight = 0x08;  // current backlight bit, OR'd into every write

  // Raw write of the PCF8574 latch.
  void writeI2C(uint8_t value) {
    Wire.beginTransmission(I2C_ADDR_LCD);
    Wire.write(value);
    Wire.endTransmission();
  }

  // Present one nibble (already in bits 7..4, RS in bit 0) and strobe EN.
  void pulseNibble(uint8_t value) {
    value |= backlight;
    writeI2C(value | 0x04);   // EN high
    delayMicroseconds(1);     // EN pulse width — inherent to the protocol
    writeI2C(value & ~0x04);  // EN low: LCD latches the nibble here
  }

  // Send a full byte as two nibbles; rs = 0x01 for character data, 0 for command.
  void sendByte(uint8_t value, uint8_t rs) {
    pulseNibble((value & 0xF0) | rs);
    pulseNibble(((value << 4) & 0xF0) | rs);
  }

  void command(uint8_t value) {
    sendByte(value, 0);
  }

  // HD44780 cold-start dance: three 0x30 writes force 8-bit mode from any
  // power-up state, then 0x20 drops to 4-bit; then the usual function set.
  void begin() {
    delay(50);
    pulseNibble(0x30);
    delay(5);
    pulseNibble(0x30);
    delay(5);
    pulseNibble(0x30);
    delay(1);
    pulseNibble(0x20);
    command(0x28);  // 4-bit, 2-line mode (drives all 4 rows on a 20x4)
    command(0x0C);  // display on, cursor off, blink off
    command(0x06);  // entry mode: increment, no shift
    clear();
  }

  void clear() {
    command(0x01);
    delay(2);
  }  // setup()/BIST only (delay)

  // DDRAM quirk: 20x4 rows start at 0x00, 0x40, 0x14, 0x54 (rows interleave).
  void setCursor(uint8_t row, uint8_t col) {
    static const uint8_t ROW_ADDR[4] = { 0x00, 0x40, 0x14, 0x54 };
    command(0x80 | (ROW_ADDR[row & 3] + col));
  }

  void print(uint8_t row, uint8_t col, const char* text) {
    setCursor(row, col);
    while (*text) sendByte((uint8_t)*text++, 0x01);
  }

  void on() {
    command(0x0C);
  }
  void off() {
    command(0x08);
  }
  void backlightOn() {
    backlight = 0x08;
    writeI2C(backlight);
  }  // refresh latch, no EN pulse
  void backlightOff() {
    backlight = 0x00;
    writeI2C(backlight);
  }
};

// ─────────────────────────────────────────────────────────────────────────────
// Snapshots
//
// `inputs` is filled once at the top of every loop() pass so all three
// controllers see the same world this tick. `outputs` is what the controllers
// produce; the displays render exclusively from these two structs. Each
// carries the millis() timestamp of its last update.
// ─────────────────────────────────────────────────────────────────────────────
struct InputSnapshot {
  uint32_t ms = 0;
  uint16_t airX10 = 0;     // air supply, PSI x10
  uint16_t n2LowX100 = 0;  // low-pressure N2, PSI x100
  uint16_t n2HighX10 = 0;  // high-pressure N2, PSI x10
  bool tbs = false, tob = false;
} inputs;

struct OutputSnapshot {
  uint32_t ms = 0;
  bool left = false, right = false, flush = false, ssr = false;  // LRFS
  uint16_t n2pctX100 = 0;                                        // nitrogen %, x100, ceiling 9999 (= 99.99%)
  bool n2Valid = false;                                          // false until the first completed sample-cycle
} outputs;

// Drive a physical output pin and record its state in the output snapshot.
// Every output change in the sketch passes through this single funnel.
void driveOutput(uint8_t pin, bool on, bool& snapshotField) {
  digitalWrite(pin, on ? HIGH : LOW);
  snapshotField = on;
  outputs.ms = millis();  // update timestamp
}

// scalePressure — adapted to file-scope constants.
// Clamps the raw ADC count to the transducer's live window (0.5–4.5 V) and
// maps linearly onto 0..fullScale in the sensor's fixed-point representation.
static uint16_t scalePressure(int raw, uint16_t fullScale) {
  const int clamped = constrain(raw, RAW_MIN, RAW_MAX);
  return static_cast<uint16_t>(map(clamped, RAW_MIN, RAW_MAX, 0, static_cast<int>(fullScale)));
}

// Rollover-safe deadline test: the signed difference of unsigned millis()
// values stays correct across the ~49.7-day wrap for spans under 2^31 ms.
static bool deadlineReached(uint32_t now, uint32_t deadline) {
  return (int32_t)(now - deadline) >= 0;
}

// ─────────────────────────────────────────────────────────────────────────────
// Shared transition log
//
// One console line per state transition:
//     now+delta NAME OLD->NEW next:deadline
// `delta` is the time since this same controller's previous line; each
// controller has its own slot in lastLogMs, and the same ID indexes the
// static CONTROLLER_NAME table, so callers pass only the ID. State names are
// the short forms (also used on the LCD) — one name table per controller.
// `deadline` 0 means no timed transition is pending and prints as "next:-".
// ─────────────────────────────────────────────────────────────────────────────
const char* const CONTROLLER_NAME[3] = { "TWR", "CMP", "O2" };  // IDs 0, 1, 2
uint32_t lastLogMs[3] = { 0, 0, 0 };

void logTransition(uint8_t id, const char* fromState, const char* toState, uint32_t deadline) {
  uint32_t now = millis();  // should this be the latest timestamp from the output snapshot?
  Serial.print(now);
  Serial.print("+");
  Serial.print(now - lastLogMs[id]);
  lastLogMs[id] = now;
  Serial.print(' ');
  Serial.print(CONTROLLER_NAME[id]);
  Serial.print(' ');
  Serial.print(fromState);
  Serial.print("->");
  Serial.print(toState);
  Serial.print(" next:");
  if (deadline) Serial.println(deadline);
  else Serial.println('-');
}

enum Controller : uint8_t {
  TWR = 0,
  CMP = 1,
  O2C = 2
};

// ─────────────────────────────────────────────────────────────────────────────
// Tower controller (ID 0)
//
// Theory: PSA tower cycling. The duty cycle alternates which tower fills
// (LEFT, RIGHT) with a short overlap phase where both valves are open
// (LEFT_BOTH, RIGHT_BOTH) so the supply is never dead-headed. Phase changes
// are purely timed (fill / overlap durations). Two safety conditions are
// re-checked on EVERY tick from any running state and jump straight to
// DISABLED, bypassing any pending deadline: air supply too low to run
// (below airLowOff), or the N2 storage already full (above n2HighOff).
// Leaving DISABLED requires the hysteresis-recovery side of both thresholds —
// air back above airLowOn AND storage drawn down below n2HighOn — and TBS on,
// so pressure alone can never start the machine while it is switched off.
// ─────────────────────────────────────────────────────────────────────────────
namespace Tower {
enum State : uint8_t { DISABLED,
                       LEFT,
                       LEFT_BOTH,
                       RIGHT,
                       RIGHT_BOTH };
const char* const STATE_NAME[] = { "OFF", "L", "LB", "R", "RB" };

State state = DISABLED;
uint32_t deadline = 0;

// Set both valves in one call; every DISABLED entry goes through here.
void valves(bool leftOpen, bool rightOpen) {
  driveOutput(LEFT_TOWER_VALVE_PIN, leftOpen, outputs.left);
  driveOutput(RIGHT_TOWER_VALVE_PIN, rightOpen, outputs.right);
}

// The only place state/deadline are assigned: log first, then commit.
void transition(State newState, uint32_t newDeadline) {
  logTransition(TWR, STATE_NAME[state], STATE_NAME[newState], newDeadline);
  state = newState;
  deadline = newDeadline;
}

void setup() {
  valves(false, false);
}  // safe state at boot
void disable() {
  valves(false, false);
  if (state != DISABLED) transition(DISABLED, 0);
}
void enable() {}  // state is already DISABLED; cycling restarts through the
                  // normal DISABLED-exit rule once pressures permit

void update() {
  const uint32_t now = inputs.ms;

  // Safety: checked every tick from any running state; bypasses the deadline.
  if (state != DISABLED && (inputs.airX10 < airLowOff || inputs.n2HighX10 > n2HighOff)) {
    valves(false, false);
    transition(DISABLED, 0);
    return;
  }

  switch (state) {
    case DISABLED:
      // Exit needs TBS on plus the recovery side of both thresholds.
      if (inputs.tbs && inputs.airX10 > airLowOn && inputs.n2HighX10 < n2HighOn) {
        valves(true, false);
        transition(LEFT, now + towerFillTime);
      }
      break;
    case LEFT:  // left filling → open right too (overlap)
      if (deadlineReached(now, deadline)) {
        driveOutput(RIGHT_TOWER_VALVE_PIN, true, outputs.right);
        transition(LEFT_BOTH, now + towerOverlapTime);
      }
      break;
    case LEFT_BOTH:  // overlap done → right alone fills
      if (deadlineReached(now, deadline)) {
        driveOutput(LEFT_TOWER_VALVE_PIN, false, outputs.left);
        transition(RIGHT, now + towerFillTime);
      }
      break;
    case RIGHT:  // right filling → open left too (overlap)
      if (deadlineReached(now, deadline)) {
        driveOutput(LEFT_TOWER_VALVE_PIN, true, outputs.left);
        transition(RIGHT_BOTH, now + towerOverlapTime);
      }
      break;
    case RIGHT_BOTH:  // overlap done → left alone fills; the cycle repeats
      if (deadlineReached(now, deadline)) {
        driveOutput(RIGHT_TOWER_VALVE_PIN, false, outputs.right);
        transition(LEFT, now + towerFillTime);
      }
      break;
  }
}
}

// ─────────────────────────────────────────────────────────────────────────────
// N2 compressor controller (ID 1)
//
// Theory: pure input-driven hysteresis at both ends of the storage system —
// no deadlines anywhere. RUNNING drives the SSR. Two stop reasons: the
// low-pressure feed dips below n2LowOff (the compressor would starve its
// own supply), or the high-pressure tank passed n2HighOff (full). Each stop
// remembers WHY it stopped, and only its own recovery threshold restarts it:
// STOPPED_LOW waits for the feed to rebuild above n2LowOn; STOPPED_HIGH waits
// for the tank to draw down below n2HighOn.
// ─────────────────────────────────────────────────────────────────────────────
namespace Compressor {
enum State : uint8_t { DISABLED,
                       RUNNING,
                       STOPPED_LOW,
                       STOPPED_HIGH };
const char* const STATE_NAME[] = { "OFF", "ON", "LO", "HI" };

State state = DISABLED;

void ssr(bool on) {
  driveOutput(SSR_PIN, on, outputs.ssr);
}

void transition(State newState) {
  logTransition(CMP, STATE_NAME[state], STATE_NAME[newState], 0);  // never timed
  state = newState;
}

void setup() {
  ssr(false);
}
void disable() {
  ssr(false);
  if (state != DISABLED) transition(DISABLED);
}

// Enable lands in STOPPED_LOW so the low-side restart rule (feed above
// n2LowOn) governs the first SSR-on; rules re-evaluate on the next tick.
void enable() {
  transition(STOPPED_LOW);
}

void update() {
  switch (state) {
    case RUNNING:
      if (inputs.n2LowX100 < n2LowOff) {
        ssr(false);
        transition(STOPPED_LOW);
      } else if (inputs.n2HighX10 > n2HighOff) {
        ssr(false);
        transition(STOPPED_HIGH);
      }
      break;
    case STOPPED_LOW:
      if (inputs.n2LowX100 > n2LowOn) {
        ssr(true);
        transition(RUNNING);
      }
      break;
    case STOPPED_HIGH:
      if (inputs.n2HighX10 < n2HighOn) {
        ssr(true);
        transition(RUNNING);
      }
      break;
    case DISABLED: break;  // only enable() leaves this state
  }
}
}

// ─────────────────────────────────────────────────────────────────────────────
// O2 controller (ID 2)
//
// Theory: every O2SampleInterval (60 s, measured cycle-start to cycle-start)
// the controller opens the flush valve for O2FlushTime to draw fresh product
// gas past the sensor, closes it, then takes O2SampleCount readings spaced
// O2SampleTime apart and stores the rounded average — converted to N2% — in
// the output snapshot. The 250 ms per-sample timing reuses the state
// machine's own deadline, re-armed inside SAMPLING with a sample counter;
// re-arms are not transitions and are not logged. The whole burst is 2.5 s,
// so the FLUSHING->SAMPLING and SAMPLING->WAITING lines bracket it usefully.
//
// Comm handling: the DFRobot library surfaces no errors, so a raw I2C ping
// (zero-length transmission) guards every read. Any failure logs, closes the
// flush valve, and parks in ERROR until TBS off→on re-enters UNKNOWN, which
// re-verifies communication (pinging every 1 s until the sensor answers).
// ─────────────────────────────────────────────────────────────────────────────
namespace O2 {
enum State : uint8_t { UNKNOWN,
                       FLUSHING,
                       SAMPLING,
                       WAITING,
                       ERROR,
                       DISABLED };
const char* const STATE_NAME[] = { "??", "F", "S", "W", "E", "OFF" };

State state = DISABLED;
uint32_t deadline = 0;
uint32_t cycleStart = 0;  // anchor: next cycle begins at cycleStart + O2SampleInterval
uint32_t sampleSum = 0;
uint8_t sampleNum = 0;

void flushValve(bool open) {
  driveOutput(O2_FLUSH_VALVE_PIN, open, outputs.flush);
}

void transition(State newState, uint32_t newDeadline) {
  logTransition(O2C, STATE_NAME[state], STATE_NAME[newState], newDeadline);
  state = newState;
  deadline = newDeadline;
}

// Zero-length I2C transaction: true iff the sensor ACKs its address.
bool sensorPresent() {
  Wire.beginTransmission(I2C_ADDR_O2);
  return Wire.endTransmission() == 0;
}

// Read O2 concentration as integer x100. Float exists only here, at the
// sensor-library boundary, and is rounded into fixed-point immediately.
bool readO2PercentX100(uint16_t& value) {
  if (!sensorPresent()) return false;
  float pct = gasO2.readGasConcentrationPPM();  // %vol for the O2 SEN0465
  if (pct < 0) pct = 0;
  value = (uint16_t)(pct * 100.0f + 0.5f);
  return true;
}

void fail() {
  Serial.print('[');
  Serial.print(millis());
  Serial.println("] O2 sensor communication FAILURE");
  flushValve(false);
  transition(ERROR, 0);  // persists until TBS off→on re-enters UNKNOWN
}

// Open the flush valve and anchor the new cycle's timing.
void beginSampleCycle(uint32_t now) {
  flushValve(true);
  cycleStart = now;
  transition(FLUSHING, now + O2FlushTime);
}

void setup() {
  flushValve(false);
}
void disable() {
  flushValve(false);
  if (state != DISABLED) transition(DISABLED, 0);
}
void enable() {
  transition(UNKNOWN, millis());
}  // deadline = now: try comm immediately

void update() {
  const uint32_t now = inputs.ms;
  switch (state) {
    case UNKNOWN:
      if (deadlineReached(now, deadline)) {
        if (sensorPresent() && gasO2.begin()) {
          gasO2.changeAcquireMode(gasO2.PASSIVITY);  // we poll; the sensor doesn't stream
          beginSampleCycle(now);
        } else {
          deadline = now + 1000;  // retry ping every 1 s (re-arm, not logged)  TODO: Parameterize?
        }
      }
      break;
    case FLUSHING:
      if (deadlineReached(now, deadline)) {
        flushValve(false);
        uint16_t value;
        if (!readO2PercentX100(value)) {
          fail();
          break;
        }
        sampleSum = value;
        sampleNum = 1;  // first sample taken on entry
        transition(SAMPLING, now + O2SampleTime);
      }
      break;
    case SAMPLING:
      if (deadlineReached(now, deadline)) {
        uint16_t value;
        if (!readO2PercentX100(value)) {
          fail();
          break;
        }
        sampleSum += value;
        if (++sampleNum >= O2SampleCount) {
          // Rounded average, then N2% = 100.00 − O2%, ceiling 99.99.
          uint16_t avg = (uint16_t)((sampleSum + sampleNum / 2) / sampleNum);  // round
          uint16_t n2 = (avg >= 10000) ? 0 : (uint16_t)(10000 - avg);
          if (n2 > 9999) n2 = 9999;
          outputs.n2pctX100 = n2;
          outputs.n2Valid = true;
          outputs.ms = now;
          transition(WAITING, cycleStart + O2SampleInterval);
        } else {
          deadline = now + O2SampleTime;  // per-sample re-arm, not logged
        }
      }
      break;
    case WAITING:
      if (deadlineReached(now, deadline)) beginSampleCycle(now);
      break;
    case ERROR:
    case DISABLED:
      break;  // inert; only enable()/disable() move us
  }
}
}

// ─────────────────────────────────────────────────────────────────────────────
// Displays — Layout C; every field has a fixed home; change-only writes
//
//   ....5....0....5....0
//   AIR 123.4  TWR  LB
//   N2L 12.34  CMP  ON
//   N2H 123.4  O2   S
//   N2% 99.99  LRFS 1001
//
// Static labels are painted once after BIST. Each dynamic field is formatted
// into a small buffer and compared with its cached previous text; only a
// changed field is written, at its fixed (row, col). This is what makes
// "refresh every tick, no rate limit" cheap: an idle screen costs zero I2C.
// ─────────────────────────────────────────────────────────────────────────────
struct LcdFieldCache {
  char air[6], n2Low[6], n2High[6], n2Pct[6];    // value fields, 5 chars each
  char tower[5], compressor[5], o2[5], bits[5];  // state fields, 4 chars each
} lcdCache;                                      // zero-initialized → every field differs on the first pass

void lcdWriteField(uint8_t row, uint8_t col, const char* text, char* cache) {
  if (strcmp(text, cache) != 0) {
    strcpy(cache, text);
    lcd.print(row, col, text);
  }
}

void formatX10(char* buf, uint16_t value) {
  snprintf(buf, 6, "%3u.%u", value / 10, value % 10);
}  // nnn.n
void formatX100(char* buf, uint16_t value) {
  snprintf(buf, 6, "%2u.%02u", value / 100, value % 100);
}  // nn.nn

void lcdLabels() {
  lcd.print(0, 0, "AIR");
  lcd.print(0, 11, "TWR");
  lcd.print(1, 0, "N2L");
  lcd.print(1, 11, "CMP");
  lcd.print(2, 0, "N2H");
  lcd.print(2, 11, "O2");
  lcd.print(3, 0, "N2%");
  lcd.print(3, 11, "LRFS");
}

void updateDisplays() {
  char buf[8];

  // Left column: the four pressure/percentage values at column 4.
  formatX10(buf, inputs.airX10);
  lcdWriteField(0, 4, buf, lcdCache.air);
  formatX100(buf, inputs.n2LowX100);
  lcdWriteField(1, 4, buf, lcdCache.n2Low);
  formatX10(buf, inputs.n2HighX10);
  lcdWriteField(2, 4, buf, lcdCache.n2High);
  if (outputs.n2Valid) formatX100(buf, outputs.n2pctX100);
  else strcpy(buf, "--.--");
  lcdWriteField(3, 4, buf, lcdCache.n2Pct);

  // Right column: controller states (space-padded to 4 so a shorter name
  // fully overwrites a longer one) and the LRFS output bitmap, at column 16.
  snprintf(buf, 5, "%-4s", Tower::STATE_NAME[Tower::state]);
  lcdWriteField(0, 16, buf, lcdCache.tower);
  snprintf(buf, 5, "%-4s", Compressor::STATE_NAME[Compressor::state]);
  lcdWriteField(1, 16, buf, lcdCache.compressor);
  snprintf(buf, 5, "%-4s", O2::STATE_NAME[O2::state]);
  lcdWriteField(2, 16, buf, lcdCache.o2);
  buf[0] = '0' + outputs.left;
  buf[1] = '0' + outputs.right;
  buf[2] = '0' + outputs.flush;
  buf[3] = '0' + outputs.ssr;
  buf[4] = 0;
  lcdWriteField(3, 16, buf, lcdCache.bits);

  // LED: N2% when enabled; dashes until the first reading (or in ERROR);
  // blank when the system is switched off.
  if (!inputs.tbs) led.blank();
  else if (outputs.n2Valid) led.show(outputs.n2pctX100, 1);  // dot at digit 1 → nn.nn
  else led.dashes(1);
}

// ─────────────────────────────────────────────────────────────────────────────
// BIST — runs once inside setup(); delay() and TOB spin-waits permitted here.
//
// The LED shows the step number in HEX except during the LED test (step 3);
// the LCD shows the step number except during the LCD test (step 4). The
// operator judges pass/fail by eye and ear and presses TOB to advance;
// nothing is recorded. Both displays are driven through the same
// mini-libraries the production code uses, so the display path itself is
// under test throughout.
// ─────────────────────────────────────────────────────────────────────────────
bool tobPressed() {
  return digitalRead(TOB_PIN) == LOW;
}

// Wait out the release (plus a settle) so one press advances exactly one step.
void tobWaitRelease() {
  while (tobPressed()) {}
  delay(30);
}

// Block for the press itself: spin while OFF until ON, then absorb the release.
void waitForTob() {
  while (!tobPressed()) {}
  tobWaitRelease();
}

// A delay that aborts early on TOB; returns true if TOB ended it.
bool delayPoll(uint32_t ms) {
  uint32_t start = millis();
  while (millis() - start < ms) {
    if (tobPressed()) {
      tobWaitRelease();
      return true;
    }
    delay(5);
  }
  return false;
}

// Announce a step on console / LED / LCD (suppressed on the device under test).
void bistBanner(uint8_t step, const char* name, bool onLed = true, bool onLcd = true) {
  Serial.print("BIST ");
  Serial.print(step, HEX);
  Serial.print(": ");
  Serial.println(name);
  if (onLed) led.showHex(step);
  if (onLcd) {
    lcd.clear();
    char buf[21];
    snprintf(buf, 21, "BIST %X %s", step, name);
    lcd.print(0, 0, buf);
  }
}

// One output's test: toggle at ~2 Hz so the operator HEARS the valve open and
// close; every change is logged and mirrored on the LCD; TOB moves on. The
// output is always left OFF afterward.
void bistOutputTest(uint8_t step, const char* name, uint8_t pin) {
  bistBanner(step, name);
  bool on = false;
  for (;;) {
    on = !on;
    digitalWrite(pin, on ? HIGH : LOW);
    Serial.print(name);
    Serial.println(on ? " ON" : " OFF");
    lcd.print(1, 0, on ? "ON " : "OFF");
    if (delayPoll(250)) break;  // 250 ms per half-cycle ≈ 2 Hz open/close
  }
  digitalWrite(pin, LOW);
}

void bist() {

  // BIST 0 — banner; prove TOB works before anything depends on it.
  Serial.println("Each BIST step terminates when TOB (The Other Button) is pressed...\n");
  bistBanner(0, "press TOB");
  waitForTob();

  // BIST 1 — TBS: echo the live switch state; log only on change.
  bistBanner(1, "Enable/Disable TBS (The Black Switch) a few times, then press TOB");
  {
    bool previous = !(digitalRead(TBS_PIN) == LOW);  // inverted so the first pass always prints
    for (;;) {
      bool tbsOn = (digitalRead(TBS_PIN) == LOW);
      if (tbsOn != previous) {
        previous = tbsOn;
        Serial.println(tbsOn ? "TBS ON" : "TBS OFF");
        lcd.print(1, 0, tbsOn ? "TBS ON " : "TBS OFF");
      }
      if (tobPressed()) {
        tobWaitRelease();
        break;
      }
    }
  }

  // BIST 2 — I2C scan, before the LED/LCD tests since both need the bus.
  //     Scan once; log responsive addresses only; wait for TOB.
  bistBanner(2, "I2C scan");
  Serial.println("responders:");
  uint16_t addrFound = 0;
  for (uint8_t addr = 8; addr < 0x78; addr++) {
    Serial.print(".");
    Serial.flush();
    Wire.beginTransmission(addr);
    if (Wire.endTransmission() == 0) {
      Serial.print(" 0x");
      Serial.print(addr, HEX);
      Serial.flush();
      addrFound++;
    }
  }
  Serial.print("\nScan found ");
  Serial.print(addrFound);
  Serial.println(" device(s). Waiting for TOB...");
  waitForTob();

  // BIST 3 — LED pattern test (LED is the device under test; step # on LCD only).
  bistBanner(3, "LED test", false, true);
  for (;;) {
    bool quit = false;
    for (uint16_t d = 0; d <= 9 && !quit; d++) {
      led.show(d * 1111);
      quit = delayPoll(200);
    }  // 0000..9999
    for (int8_t p = 0; p < 4 && !quit; p++) {
      led.show(8888, p);
      quit = delayPoll(350);
    }  // walk the dot
    if (!quit) {
      led.off();
      quit = delayPoll(400);
      led.on();
    }  // display on/off
    if (quit) break;
  }
  led.on();


  // BIST 4 — LCD pattern test (LCD is the device under test; step # on LED only).
  led.showHex(4);
  Serial.println("BIST 4: LCD test");
  for (;;) {
    bool quit = false;
    lcd.backlightOff();
    quit = delayPoll(500);
    if (!quit) {
      lcd.backlightOn();
      quit = delayPoll(500);
    }
    if (!quit) {
      lcd.off();
      quit = delayPoll(500);
    }
    if (!quit) { lcd.on(); }
    if (!quit) {  // fill every one of the 80 cells
      char row[21];
      memset(row, '#', 20);
      row[20] = 0;
      for (uint8_t r = 0; r < 4; r++) lcd.print(r, 0, row);
      quit = delayPoll(1500);
    }
    if (!quit) {
      lcd.clear();
      quit = delayPoll(500);
    }
    if (quit) break;
  }
  lcd.backlightOn();
  lcd.on();

  // BIST 5 — PSI inputs: log/refresh the first readings, then only changes.
  bistBanner(5, "PSI inputs");
  {
    pinMode(SUPPLY_PRESSURE_PIN, INPUT);
    pinMode(LOW_N2_PIN, INPUT);
    pinMode(HIGH_N2_PIN, INPUT);
    uint16_t prevAir = 0xFFFF, prevLow = 0xFFFF, prevHigh = 0xFFFF;  // impossible → first pass prints all
    for (;;) {
      uint16_t air = scalePressure(analogRead(SUPPLY_PRESSURE_PIN), AIR_FULLSCALE_X10);
      uint16_t low = scalePressure(analogRead(LOW_N2_PIN), N2LOW_FULLSCALE_X100);
      uint16_t high = scalePressure(analogRead(HIGH_N2_PIN), N2HIGH_FULLSCALE_X10);
      char buf[21];
      if (air != prevAir) {
        prevAir = air;
        snprintf(buf, 21, "AIR %3u.%u", air / 10, air % 10);
        lcd.print(1, 0, buf);
        Serial.println(buf);
      }
      if (low != prevLow) {
        prevLow = low;
        snprintf(buf, 21, "N2L %2u.%02u", low / 100, low % 100);
        lcd.print(2, 0, buf);
        Serial.println(buf);
      }
      if (high != prevHigh) {
        prevHigh = high;
        snprintf(buf, 21, "N2H %3u.%u", high / 10, high % 10);
        lcd.print(3, 0, buf);
        Serial.println(buf);
      }
      if (delayPoll(200)) break;
    }
  }

  // BIST 6 — O2 sensor: show the first reading, then only changes.
  bistBanner(6, "O2 sensor");
  {
    bool gasOk = gasO2.begin();
    if (gasOk) gasO2.changeAcquireMode(gasO2.PASSIVITY);
    Serial.println(gasOk ? "SEN0465 begin OK" : "SEN0465 begin FAILED");
    uint16_t previous = 0xFFFF;  // impossible value → first reading always prints
    for (;;) {
      char buf[21];
      if (gasOk) {
        uint16_t o2 = (uint16_t)(gasO2.readGasConcentrationPPM() * 100.0f + 0.5f);
        if (o2 != previous) {
          previous = o2;
          snprintf(buf, 21, "O2 %2u.%02u", o2 / 100, o2 % 100);
          lcd.print(1, 0, buf);
          Serial.println(buf);
        }
      } else if (previous != 0) {
        previous = 0;
        strcpy(buf, "O2: no response");
        lcd.print(1, 0, buf);
        Serial.println(buf);
      }
      if (delayPoll(500)) break;
    }
  }

  // BIST 7..A — one test per output, ~2 Hz audible toggling, TOB to advance.
  bistOutputTest(0x7, "LEFT valve", LEFT_TOWER_VALVE_PIN);
  bistOutputTest(0x8, "RIGHT valve", RIGHT_TOWER_VALVE_PIN);
  bistOutputTest(0x9, "Flush valve", O2_FLUSH_VALVE_PIN);
  bistOutputTest(0xA, "SSR", SSR_PIN);

  Serial.println("BIST complete");
}

// ─────────────────────────────────────────────────────────────────────────────
// setup / loop
// ─────────────────────────────────────────────────────────────────────────────
bool previousTbs;  // last TBS level seen by loop(); initialized once in setup()

void setup() {
  pinMode(TBS_PIN, INPUT_PULLUP);
  pinMode(TOB_PIN, INPUT_PULLUP);
  //
  pinMode(SUPPLY_PRESSURE_PIN, INPUT);
  pinMode(LOW_N2_PIN, INPUT);
  pinMode(HIGH_N2_PIN, INPUT);

  pinMode(LEFT_TOWER_VALVE_PIN, OUTPUT);
  pinMode(RIGHT_TOWER_VALVE_PIN, OUTPUT);
  pinMode(O2_FLUSH_VALVE_PIN, OUTPUT);
  pinMode(SSR_PIN, OUTPUT);

  // safe (de-energized) at boot

  Serial.begin(115200);
  Wire.begin();
  led.begin();
  lcd.begin();

  Tower::setup();
  Compressor::setup();
  O2::setup();

  if (runBIST) { bist(); }  // once per powerup, regardless of TBS, before loop() ever runs

  lcd.clear();
  lcdLabels();

  // Boot as if TBS were OFF: if it is actually ON, the first loop() pass sees
  // the edge, logs it, and enables every controller.
  previousTbs = false;

  // From here on: no delay(), anywhere.
}

void loop() {
  // 1 — input snapshot: read every input exactly once so all three
  //     controllers see the same world for this tick.
  inputs.ms = millis();
  inputs.tbs = (digitalRead(TBS_PIN) == LOW);  // "The Black Switch"
  inputs.tob = (digitalRead(TOB_PIN) == LOW);  // "The Other Button"
  inputs.airX10 = scalePressure(analogRead(SUPPLY_PRESSURE_PIN), AIR_FULLSCALE_X10);
  inputs.n2LowX100 = scalePressure(analogRead(LOW_N2_PIN), N2LOW_FULLSCALE_X100);
  inputs.n2HighX10 = scalePressure(analogRead(HIGH_N2_PIN), N2HIGH_FULLSCALE_X10);

  // TBS edge: disable() drops every controller to DISABLED (closing its
  // outputs); enable() re-arms them. Both edges are logged.
  if (inputs.tbs != previousTbs) {
    Serial.print('[');
    Serial.print(inputs.ms);
    Serial.println(inputs.tbs ? "] TBS OFF->ON" : "] TBS ON->OFF");
    if (inputs.tbs) {
      Tower::enable();
      Compressor::enable();
      O2::enable();
    } else {
      Tower::disable();
      Compressor::disable();
      O2::disable();
    }
    previousTbs = inputs.tbs;
  }

  // 2 — controllers: each may transition on time or inputs, writing its
  //     outputs through driveOutput into the output snapshot.
  Tower::update();
  Compressor::update();
  O2::update();

  // 3 — displays render from the snapshots; change-only writes keep an
  //     idle screen at zero I2C cost.
  updateDisplays();
}
